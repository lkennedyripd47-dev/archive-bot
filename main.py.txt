# --- MA'LUMOTLAR BAZASI (SQLite) ---
def init_db():
    conn = sqlite3.connect("arxiv.db")
    cursor = conn.cursor()
    cursor.execute('''
        CREATE TABLE IF NOT EXISTS orders (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            user_id INTEGER,
            full_name TEXT,
            phone TEXT,
            service_type TEXT,
            file_id TEXT,
            status TEXT DEFAULT "Kutilmoqda",
            payment_status TEXT DEFAULT "Tolanmagan"
        )
    ''')
    conn.commit()
    conn.close()

class AdminState(StatesGroup):

async def start_cmd(message: Message, state: FSMContext):
    await state.clear()
    await message.answer(
        f"Assalomu alaykum, {message.from_user.first_name}!\n"
        f"Nodavlat arxiv xizmati botiga xush kelibsiz.\n\n"
        f"Xizmatlardan foydalanish uchun quyidagi menyudan foydalaning:",

@router.message(ApplicationState.waiting_for_name)

@router.message(ApplicationState.waiting_for_phone)
async def process_phone(message: Message, state: FSMContext):
    phone = message.contact.phone_number if message.contact else message.text
    await state.update_data(phone=phone)
    
    await message.answer("Kerakli xizmat turini tanlang:", reply_markup=services_menu())
    await state.set_state(ApplicationState.waiting_for_service)

@router.callback_query(ApplicationState.waiting_for_service, F.data.startswith("serv_"))
async def process_service(call: CallbackQuery, state: FSMContext):
    services = {
        "serv_staj": "Ish stajini tasdiqlash",
        "serv_haq": "Ish haqi ma'lumotnomasi",
        "serv_boshqa": "Boshqa arxiv hujjatlari"
    }
    selected_service = services.get(call.data, "Noma'lum xizmat")
    await state.update_data(service_type=selected_service)
    
    await call.message.answer("Hujjatingiz nusxasini (Pasport yoki Mehnat daftarchasi) Rasm yoki PDF shaklida yuboring:")
    await state.set_state(ApplicationState.waiting_for_file)
    await call.answer()

@router.message(ApplicationState.waiting_for_file, F.photo | F.document)
async def process_file(message: Message, state: FSMContext):
    file_id = message.photo[-1].file_id if message.photo else message.document.file_id
    data = await state.get_data()
    
    # Bazaga saqlash
    conn = sqlite3.connect("arxiv.db")
    cursor = conn.cursor()
    cursor.execute(
        "INSERT INTO orders (user_id, full_name, phone, service_type, file_id) VALUES (?, ?, ?, ?, ?)",
        (message.from_user.id, data['full_name'], data['phone'], data['service_type'], file_id)
    )
    order_id = cursor.lastrowid
    conn.commit()
    conn.close()

    await message.answer(
        f"✅ Arizangiz qabul qilindi!\n\n"
        f"🆔 Ariza raqami: #{order_id}\n"
        f"👤 F.I.SH: {data['full_name']}\n"
        f"📌 Xizmat: {data['service_type']}\n\n"
        f"Tez orada arizangiz ko'rib chiqiladi.",
        reply_markup=main_menu()
    )

    # Admin (Arxivchi)ga xabar yuborish
    admin_text = (
        f"📥 **Yangi ariza tushdi! (#ID_{order_id})**\n\n"
        f"👤 Mijoz: {data['full_name']}\n"
        f"📞 Tel: {data['phone']}\n"
        f"📌 Xizmat: {data['service_type']}"
    )
    
    kb_admin = InlineKeyboardMarkup(inline_keyboard=[
        [InlineKeyboardButton(text="📄 Hujjatni tayyorlab yuborish", callback_data=f"reply_{order_id}_{message.from_user.id}")]
    ])
    
    if message.photo:
        await bot.send_photo(chat_id=ADMIN_ID, photo=file_id, caption=admin_text, reply_markup=kb_admin)
    else:
        await bot.send_document(chat_id=ADMIN_ID, document=file_id, caption=admin_text, reply_markup=kb_admin)

    await state.clear()

# --- ADMIN JAVOB BERISH MANTIQLARI ---
@router.callback_query(F.data.startswith("reply_"))
async def admin_reply_start(call: CallbackQuery, state: FSMContext):
    _, order_id, user_id = call.data.split("_")
    await state.update_data(target_order_id=order_id, target_user_id=user_id)
    await call.message.answer(f"#{order_id}-sonli ariza uchun tayyor tayyorlangan hujjatni (PDF yoki Rasm shaklida) yuboring:")
    await state.set_state(AdminState.waiting_for_reply_file)
    await call.answer()

@router.message(AdminState.waiting_for_reply_file, F.photo | F.document)
async def admin_send_file(message: Message, state: FSMContext):
    data = await state.get_data()
    target_user_id = int(data['target_user_id'])
    target_order_id = data['target_order_id']
    
    file_id = message.photo[-1].file_id if message.photo else message.document.file_id
    
    # Mijozga yuborish
    caption = f"🎉 **Arizangiz tayyor bo'ldi!**\n\n🆔 Ariza raqami: #{target_order_id}\nNodavlat arxiv xizmatimizdan foydalanganingiz uchun rahmat!"
    if message.photo:
        await bot.send_photo(chat_id=target_user_id, photo=file_id, caption=caption)
    else:
        await bot.send_document(chat_id=target_user_id, document=file_id, caption=caption)
        
    # Bazada statusni yangilash
    conn = sqlite3.connect("arxiv.db")
    cursor = conn.cursor()
    cursor.execute("UPDATE orders SET status = 'Bajarildi' WHERE id = ?", (target_order_id,))
    conn.commit()
    conn.close()

    await message.answer(f"✅ #{target_order_id}-sonli ariza bo'yicha javob mijozga muvaffaqiyatli yuborildi!")
    await state.clear()
